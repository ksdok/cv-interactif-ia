✅ Automatic Detection
   • Detects browser language (navigator.language)
   • French speakers 🇫🇷 → French UI by default
   • Everyone else → English UI 🇬🇧
   • Works with: fr-FR, fr-CA, en-US, en-GB, etc.

✅ Manual Toggle Button
   • Flag emoji button in header (next to dark mode)
   • Click to switch: 🇫🇷 ↔️ 🇬🇧
   • No page reload needed
   • Instant language change

✅ Persistent Memory
   • Saves preference in browser localStorage
   • Remembers choice across sessions
   • Survives browser restart

┌──────────────────────────────────┐

💻 USAGE IN COMPONENTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

import { useLanguage } from '@/lib/LanguageContext'
import { getTranslation } from '@/lib/translations'

export default function MyComponent() {
  const { language, setLanguage } = useLanguage()
  const t = (key: string) => getTranslation(language, key)

  return (
    <>
      <h1>{t('about.title')}</h1>
      <p>{t('about.description')}</p>
      <button onClick={() => setLanguage('fr')}>
        Switch to French
      </button>
    </>
  )
}
